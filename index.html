-- Таблицы CREATE TABLE objects ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), name TEXT NOT NULL, created_at TIMESTAMPTZ DEFAULT NOW() ); CREATE TABLE periods ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), type TEXT NOT NULL CHECK (type IN ('day', 'month')), date DATE NOT NULL, label TEXT NOT NULL, created_at TIMESTAMPTZ DEFAULT NOW() ); CREATE TABLE indicators ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), name TEXT NOT NULL, unit TEXT NOT NULL DEFAULT '—', created_at TIMESTAMPTZ DEFAULT NOW() ); CREATE TABLE data ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), object_id UUID NOT NULL REFERENCES objects(id) ON DELETE CASCADE, period_id UUID NOT NULL REFERENCES periods(id) ON DELETE CASCADE, indicator_id UUID NOT NULL REFERENCES indicators(id) ON DELETE CASCADE, value NUMERIC, created_at TIMESTAMPTZ DEFAULT NOW(), updated_at TIMESTAMPTZ DEFAULT NOW(), UNIQUE(object_id, period_id, indicator_id) ); CREATE TABLE profiles ( id UUID PRIMARY KEY REFERENCES auth.users(id) ON DELETE CASCADE, role TEXT NOT NULL DEFAULT 'user' CHECK (role IN ('user', 'admin')), created_at TIMESTAMPTZ DEFAULT NOW() ); -- Включаем RLS (можно временно отключить для теста, но лучше оставить) ALTER TABLE objects ENABLE ROW LEVEL SECURITY; ALTER TABLE periods ENABLE ROW LEVEL SECURITY; ALTER TABLE indicators ENABLE ROW LEVEL SECURITY; ALTER TABLE data ENABLE ROW LEVEL SECURITY; ALTER TABLE profiles ENABLE ROW LEVEL SECURITY; -- Даём доступ для всех аутентифицированных пользователей (для простоты) CREATE POLICY "Allow all for authenticated" ON objects FOR ALL USING (auth.role() = 'authenticated'); CREATE POLICY "Allow all for authenticated" ON periods FOR ALL USING (auth.role() = 'authenticated'); CREATE POLICY "Allow all for authenticated" ON indicators FOR ALL USING (auth.role() = 'authenticated'); CREATE POLICY "Allow all for authenticated" ON data FOR ALL USING (auth.role() = 'authenticated'); CREATE POLICY "Allow all for authenticated" ON profiles FOR ALL USING (auth.role() = 'authenticated');
